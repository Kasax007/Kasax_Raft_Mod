# Run after raft_3.txt: tools/bctest/run.sh tools/content/vessel/tests/raft_4.txt <outdir>
# Raft arrival on login: reopen the world kept by raft_3.txt (the raft drifts in circles, the player stood on its deck when
# the world was saved). The player must be on the deck after loading; then the raft is stopped and anchored at the helm.
open arrivalsave
rd 5
gui 2
wait 20
settle 120
swait 40
log LOGGED_IN
count raft:raftdim
where
cmd data get entity @e[type=raft:vessel,limit=1] Pos
cmd data get entity @e[type=raft:vessel,limit=1] Rotation
cmd data get entity @p Pos
third on
wait 5 render
shot 70_after_login
third off
swait 40
where
cmd data get entity @e[type=raft:vessel,limit=1] Pos
cmd data get entity @e[type=raft:vessel,limit=1] Rotation
log ANCHOR_AT_HELM
cmd data merge entity @e[type=raft:vessel,limit=1] {Autopilot:{Ticks:0}}
swait 200
# where the helm is: stand on top of it (px/pz = its centre; the deck is at y 100, the helm centre at 100.5)
cmd execute as @e[type=raft:vessel,limit=1] at @s rotated as @s positioned ^ ^2.4 ^-3 run tp @p ~ ~ ~
swait 5
where
# stand on the deck in front of the helm, to one side (not edge-on to the thin wheel)
cmd execute as @e[type=raft:vessel,limit=1] at @s rotated as @s positioned ^ ^1.5 ^-3 run tp @p ^1.2 ^-0.4 ^1.0
swait 10
face {px} 100.6 {pz}
swait 5
wait 3 render
shot aim_helm
sneakuse
swaitfor 600 execute in raft:raftdim unless entity @e[type=raft:vessel]
count raft:raftdim
find raft:helm 48
find minecraft:chest 48
cmd execute in raft:raftdim positioned {fx} {fy} {fz} run data get block ~ ~ ~ Items
wait 10 render
shot 71_anchored
log DONE
