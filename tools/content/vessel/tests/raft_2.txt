# Run after raft_1.txt: tools/bctest/run.sh tools/content/vessel/tests/raft_2.txt <outdir>
# Raft building with a shipyard, run 4: reopen the world kept by run 3 (the raft was at sea, its blocks in plot 0),
# check that everything is still there, then drop anchor and check the blocks in the world.
open shipyardsave
rd 5
gui 2
wait 20
settle 120
swait 40
log RELOADED
count raft:raftdim
where
cmd data get entity @e[type=raft:vessel,limit=1] Pos
cmd data get entity @e[type=raft:vessel,limit=1] Plot
cmd execute in raft:raftdim if block 28000013 217 28000013 minecraft:chest
cmd execute in raft:raftdim run data get block 28000013 217 28000013 Items
cmd execute in raft:raftdim run data get block 28000009 217 28000010 Items
cmd execute in raft:raftdim if block 28000014 217 28000015 minecraft:redstone_lamp[lit=true]
third on
wait 10 render
shot 40_reloaded
# no automatic anchoring: still a raft after a while
swait 200
count raft:raftdim
# drop anchor by input: stand next to the helm (helm centre = raft pivot + ^0 ^1.5 ^-3), sneak and use it
log DROP_ANCHOR_HELM
third off
cmd execute as @e[type=raft:vessel,limit=1] at @s rotated as @s positioned ^ ^1.5 ^-3 run tp @p ^1.6 ^-0.4 ^ facing ^ ^ ^
swait 10
where
sneakuse
swait 300
count raft:raftdim
# (the same by NBT, in case the helm was not hit; nothing happens if the raft is anchored already)
log DROP_ANCHOR_NBT
cmd data merge entity @e[type=raft:vessel,limit=1] {DropAnchor:1b}
swait 300
count raft:raftdim
find minecraft:chest 48
cmd execute in raft:raftdim positioned {fx} {fy} {fz} run data get block ~ ~ ~ Items
find minecraft:furnace 48
cmd execute in raft:raftdim positioned {fx} {fy} {fz} run data get block ~ ~ ~ Items
find minecraft:wheat 48
cmd execute in raft:raftdim positioned {fx} {fy} {fz} unless block ~ ~ ~ minecraft:wheat[age=0]
find minecraft:redstone_lamp 48
cmd execute in raft:raftdim positioned {fx} {fy} {fz} if block ~ ~ ~ minecraft:redstone_lamp[lit=true]
find raft:helm 48
cmd execute in raft:raftdim if block 28000013 217 28000013 minecraft:air
wait 10 render
shot 41_anchored
log DONE
