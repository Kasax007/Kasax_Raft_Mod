# 1.21.1 port: the player leaves a sailing raft behind and comes back.
# On 1.21.1 a raft hears of its chunk no longer being tracked through the entity unload event, without a removal
# reason: up to 3.4.1 that threw an exception (server crash when the raft's chunk was left behind, and no saving of
# the world when it was closed while a raft sailed). Here: a raft sets sail (nobody at the helm, so it holds still), the
# player goes to the overworld (the raft's chunks are let go and unloaded), the server keeps running; back in the
# raft dimension the raft is there again with its blocks in the plot and the player stands on its deck. The script ends
# with the raft still sailing, so stopping the server must save the world without an exception (see latest.log).
# Run: tools/bctest/run.sh tools/port1211/tests/raft_leave.txt <outdir>
deadline 12
world raftleave
rd 4
gui 2
chat off
cmd gamerule doMobSpawning false
cmd gamerule randomTickSpeed 0
cmd gamerule raft:shark_bites false
cmd gamerule commandModificationBlockLimit 100000
cmd gamemode creative @p
cmd tp @p 0.5 201 0.5 0 30
settle 60
cmd fill -6 200 -6 6 200 6 minecraft:spruce_planks
# ---- a raft sets sail
log SET_SAIL
cmd execute in raft:raftdim run tp @p 0.5 112 0.5 0 40
wait 20
settle 200
swait 40
cmd execute in raft:raftdim run fill -24 100 -24 24 107 48 air
cmd execute in raft:raftdim run fill -24 108 -24 24 115 48 air
cmd execute in raft:raftdim run fill -24 92 -24 24 99 48 water
cmd execute in raft:raftdim run kill @e[type=!minecraft:player,distance=..90]
swait 20
cmd execute in raft:raftdim run kill @e[type=minecraft:item,distance=..90]
cmd execute in raft:raftdim run fill -3 99 -4 3 99 4 raft:raft_foundation
cmd execute in raft:raftdim run setblock 0 100 -3 raft:helm[facing=south]
cmd execute in raft:raftdim run setblock 0 100 2 raft:sail[facing=south]
cmd execute in raft:raftdim run setblock 0 101 2 raft:sail[facing=south]
swait 20
settle 60
nearblock raft:helm 48 1.3 0.5 0
swait 10
use
swaitfor 1500 execute in raft:raftdim if entity @e[type=raft:vessel]
swait 20
hold sneak 8
swait 20
check raft_sails | execute in raft:raftdim if entity @e[type=raft:vessel]
check helm_in_plot | execute in raft:raftdim if block 28000011 217 28000009 raft:helm
# ---- the player leaves: the raft's chunk is no longer tracked, then unloaded
log LEAVE
cmd execute in minecraft:overworld run tp @p 0.5 201 0.5 0 30
settle 60
swait 400
check server_still_runs | execute in minecraft:overworld if entity @p[x=0,y=200,z=0,distance=..3]
check raft_left_behind | execute in raft:raftdim unless entity @e[type=raft:vessel]
# ---- back to the raft
log BACK
cmd execute in raft:raftdim run tp @p 0.5 112 0.5 0 40
wait 20
settle 200
swaitfor 400 execute in raft:raftdim if entity @e[type=raft:vessel]
swait 40
check raft_is_back | execute in raft:raftdim if entity @e[type=raft:vessel]
check raft_still_sails | execute in raft:raftdim unless block 0 100 -3 raft:helm
check helm_still_in_plot | execute in raft:raftdim if block 28000011 217 28000009 raft:helm
# the raft holds the player again (its colliders are back): standing on the deck, not sinking into the sea
fly off
cmd execute in raft:raftdim as @e[type=raft:vessel,limit=1] at @s run tp @p ~ ~2 ~ 0 30
swait 60
check player_on_deck | execute in raft:raftdim as @p at @s if entity @s[y=99.8,dy=0.7]
where
hud off
wait 10 render
shot r01_raft_back
hud on
checks
